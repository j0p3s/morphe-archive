package com.example.util.simpletimetracker.resources;

import android.content.Context;
import android.graphics.*;
import android.graphics.drawable.BitmapDrawable;
import android.graphics.drawable.Drawable;
import android.util.Xml;
import androidx.core.graphics.PathParser;
import org.xmlpull.v1.XmlPullParser;
import java.io.File;
import java.io.FileInputStream;

public class CustomVectorLoader {
    public static Drawable load(Context context, String filename) {
        try {
            File dir = new File(context.getExternalFilesDir(null), "custom_icons");
            File file = new File(dir, filename);
            if (!file.exists()) return null;

            XmlPullParser parser = Xml.newPullParser();
            parser.setInput(new FileInputStream(file), "UTF-8");

            float vWidth = 24f, vHeight = 24f;
            Path combinedPath = new Path();

            int event = parser.getEventType();
            while (event != XmlPullParser.END_DOCUMENT) {
                if (event == XmlPullParser.START_TAG) {
                    String name = parser.getName();
                    if ("vector".equals(name)) {
                        for (int i = 0; i < parser.getAttributeCount(); i++) {
                            String attr = parser.getAttributeName(i);
                            if (attr.contains("viewportWidth")) vWidth = Float.parseFloat(parser.getAttributeValue(i));
                            if (attr.contains("viewportHeight")) vHeight = Float.parseFloat(parser.getAttributeValue(i));
                        }
                    } else if ("path".equals(name)) {
                        for (int i = 0; i < parser.getAttributeCount(); i++) {
                            if (parser.getAttributeName(i).contains("pathData")) {
                                Path p = PathParser.createPathFromPathData(parser.getAttributeValue(i));
                                if (p != null) combinedPath.addPath(p);
                            }
                        }
                    }
                }
                event = parser.next();
            }

            int size = 96; // Render size in px
            Bitmap bitmap = Bitmap.createBitmap(size, size, Bitmap.Config.ARGB_8888);
            Canvas canvas = new Canvas(bitmap);
            Matrix matrix = new Matrix();
            matrix.setScale(size / vWidth, size / vHeight);
            combinedPath.transform(matrix);

            Paint paint = new Paint(Paint.ANTI_ALIAS_FLAG);
            paint.setColor(Color.WHITE); // White base so tinting works
            paint.setStyle(Paint.Style.FILL);
            canvas.drawPath(combinedPath, paint);

            return new BitmapDrawable(context.getResources(), bitmap);
        } catch (Exception e) {
            return null;
        }
    }
}
